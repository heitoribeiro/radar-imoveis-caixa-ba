name: Publicar interface no GitHub Pages

on:
  push:
    branches: [main]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
      - uses: actions/configure-pages@v5
      - name: Preparar interface com a última base publicada
        run: |
          mkdir -p _site/data
          cp index.html styles.css enhancements.css app-v4.js app-core.js sw.js manifest.webmanifest _site/
          cp -r icons _site/
          curl -fsSL https://heitoribeiro.github.io/radar-imoveis-caixa-ba/data/imoveis-brasil.json -o _site/data/imoveis-brasil.json
          touch _site/.nojekyll
      - uses: actions/upload-pages-artifact@v4
        with:
          path: _site

  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    needs: build
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
