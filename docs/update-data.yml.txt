name: Atualizar dados CAIXA e publicar

on:
  workflow_dispatch:
  schedule:
    - cron: '5 9 * * *'

permissions:
  contents: write
  pages: write
  id-token: write

concurrency:
  group: github-pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
      - uses: actions/setup-python@v6
        with:
          python-version: '3.13'
      - name: Atualizar listas nacionais da CAIXA
        run: python scripts/update_caixa.py
      - name: Salvar histórico
        shell: bash
        run: |
          if [[ -n "$(git status --porcelain data/first-seen.json)" ]]; then
            git config user.name "github-actions[bot]"
            git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
            git add data/first-seen.json
            git commit -m "dados: atualizar histórico nacional de imóveis CAIXA"
            git push
          fi
      - uses: actions/configure-pages@v5
      - name: Preparar site com a nova base
        run: |
          mkdir -p _site/data
          cp index.html styles.css enhancements.css app-v4.js app-core.js sw.js manifest.webmanifest _site/
          cp -r icons _site/
          cp .generated-data/imoveis-brasil.json _site/data/
          touch _site/.nojekyll
      - uses: actions/upload-pages-artifact@v4
        with:
          path: _site

  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    needs: build
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
